import { Meta } from '@storybook/addon-docs/blocks';
import React from 'react';
import DSClassSystem from "./DSClassSystem";

export const CopyButton = ({ text }) => {
    const [copied, setCopied] = React.useState(false);
    
    const handleCopy = () => {
        navigator.clipboard.writeText(text);
        setCopied(true);
        setTimeout(() => setCopied(false), 2000);
    };
    
    return (
        <button
            onClick={handleCopy}
            style={{
                marginLeft: "10px",
                padding: "4px 12px",
                fontSize: "12px",
                backgroundColor: copied ? "#4caf50" : "#0066cc",
                color: "white",
                border: "none",
                borderRadius: "4px",
                cursor: "pointer",
                transition: "all 0.2s"
            }}
        >
            {copied ? "✓ Copied!" : "Copy"}
        </button>
    );
};

This documentation showcases the standardized UI Foundation classes available for use with the `isAeroDesign` prop.

## Component Classes

<div style={{marginTop: "30px"}}>
    {
        Object.entries(DSClassSystem.components).map(([className, props]) => {
            // Skip ds-popup-close as it will be shown with ds-popup-container
            if (className === 'ds-popup-close') return null;
            
            return (
                <div key={className} style={{
                    border: "1px solid #e0e0e0",
                    borderRadius: "8px",
                    padding: "20px",
                    marginBottom: "20px",
                    backgroundColor: "#f9f9f9"
                }}>
                    <div style={{
                        display: "flex",
                        alignItems: "center",
                        marginBottom: "10px"
                    }}>
                        <div style={{
                            fontFamily: "monospace",
                            fontSize: "16px",
                            fontWeight: "500",
                            color: "#0066cc"
                        }}>
                            .{className}
                        </div>
                        <CopyButton text={className} />
                        {className === 'ds-popup-container' && (
                            <>
                                <div style={{
                                    fontFamily: "monospace",
                                    fontSize: "16px",
                                    fontWeight: "500",
                                    color: "#0066cc",
                                    marginLeft: "20px"
                                }}>
                                    .ds-popup-close
                                </div>
                                <CopyButton text="ds-popup-close" />
                            </>
                        )}
                    </div>
                    <div style={{marginBottom: "10px", fontStyle: "italic", color: "#666"}}>
                        {props.description}
                        {className === 'ds-popup-container' && (
                            <span> (with close button positioning)</span>
                        )}
                    </div>
                    <div style={{
                        backgroundColor: "white",
                        padding: "15px",
                        borderRadius: "4px",
                        marginBottom: "10px",
                        border: "1px solid #ddd"
                    }}>
                        {className === 'ds-popup-container' ? (
                            <div className="ds-popup-container" style={{
                                position: "relative",
                                backgroundColor: "#f5f5f5",
                                borderRadius: "8px",
                                padding: "30px",
                                minHeight: "150px",
                                border: "2px solid #ddd"
                            }}>
                                <div className="ds-popup-close" style={{
                                    position: "absolute",
                                    top: "8px",
                                    right: "16px",
                                    width: "24px",
                                    height: "24px",
                                    padding: "6px",
                                    cursor: "pointer",
                                    zIndex: 10,
                                    display: "flex",
                                    alignItems: "center",
                                    justifyContent: "center",
                                    transition: "background-color 0.2s"
                                }}>
                                    <i className="icon_phoenix-enclose" style={{fontSize: "16px"}}/>
                                </div>
                                <h3 style={{margin: "0 0 10px 0"}}>Popup/Modal Content</h3>
                                <p style={{margin: 0, color: "#666"}}>
                                    This demonstrates the standardized close button position (top: 8px, right: 16px)
                                </p>
                            </div>
                        ) : (
                            <div className={className} style={{
                                fontFamily: props.fontFamily,
                                fontWeight: props.fontWeight,
                                letterSpacing: props.letterSpacing,
                                position: props.position,
                                width: props.width,
                                height: props.height
                            }}>
                                Sample: {className}
                            </div>
                        )}
                    </div>
                    <div style={{fontSize: "12px", color: "#666"}}>
                        <strong>Usage:</strong> {props.usage}
                        {className === 'ds-popup-container' && (
                            <div style={{marginTop: "5px"}}>
                                <strong>Close Button Usage:</strong> {DSClassSystem.components['ds-popup-close'].usage}
                            </div>
                        )}
                    </div>
                    <details style={{marginTop: "10px"}}>
                        <summary style={{cursor: "pointer", color: "#0066cc", fontSize: "12px"}}>View Properties</summary>
                        <pre style={{
                            backgroundColor: "#f5f5f5",
                            padding: "10px",
                            borderRadius: "4px",
                            fontSize: "12px",
                            marginTop: "10px",
                            overflow: "auto"
                        }}>
                            {className === 'ds-popup-container' ? 
                                JSON.stringify({
                                    'ds-popup-container': Object.entries(props).filter(([key]) => 
                                        !['description', 'usage'].includes(key)
                                    ).reduce((obj, [key, val]) => ({...obj, [key]: val}), {}),
                                    'ds-popup-close': Object.entries(DSClassSystem.components['ds-popup-close']).filter(([key]) => 
                                        !['description', 'usage'].includes(key)
                                    ).reduce((obj, [key, val]) => ({...obj, [key]: val}), {})
                                }, null, 2)
                                :
                                JSON.stringify(Object.entries(props).filter(([key]) => 
                                    !['description', 'usage'].includes(key)
                                ).reduce((obj, [key, val]) => ({...obj, [key]: val}), {}), null, 2)
                            }

## Implementation Guidelines

    backgroundColor: "#e8f4f8",

    borderRadius: "8px",
    padding: "20px",
    marginTop: "30px"
}}>
    <h3 style={{marginTop: 0, color: "#0066cc"}}>How to Use</h3>
    
    <div style={{marginBottom: "15px"}}>
        <strong>1. Import uiFoundations in your SCSS:</strong>
        <pre style={{
            backgroundColor: "#f5f5f5",
            padding: "10px",
            borderRadius: "4px",
            marginTop: "5px"
        }}>
            @import 'sass/uiFoundations/_index.scss';
        </pre>
    </div>

    <div style={{marginBottom: "15px"}}>
        <strong>2. Enable DS compliance with isAeroDesign prop:</strong>
        <pre style={{
            backgroundColor: "#f5f5f5",
            padding: "10px",
            borderRadius: "4px",
            marginTop: "5px"
        }}>
            {`<SearchFilter isAeroDesign={true} placeholder="Search..." />`}
        </pre>
    </div>

    <div>
        <strong>3. Apply DS classes directly to elements:</strong>
        <pre style={{
            backgroundColor: "#f5f5f5",
            padding: "10px",
            borderRadius: "4px",
            marginTop: "5px"
        }}>
            {`className="ds-button-secondary"`}
        </pre>
    </div>
</div>

## SCSS Files Reference

The classes are defined in the following SCSS files:
- `src/sass/uiFoundations/_page-header.scss` - Page header and button classes
- `src/sass/uiFoundations/_page-component.scss` - Component classes (dropdowns, popups, side panels)
- `src/sass/uiFoundations/_page-metrics.scss` - Metrics and KPI classes
